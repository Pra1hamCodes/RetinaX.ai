class RetinaEnsemble(nn.Module):
    def __init__(self):
        super().__init__()
        self.backbone = EfficientNet.from_pretrained('b0')
        self.attention = AttentionModule(512, heads=8)
        self.classifier = nn.Linear(1280, num_classes)
        
    def forward(self, x):
        features = self.backbone.extract_features(x)
        attn_map = self.attention(features)
        weighted = features * attn_map
        return self.classifier(weighted)
